on_actions = {

	on_daily_SOV = {
		effect = {
			set_temp_variable = {
				var = temp_SOV_shoigu_support
				value = SOV_shoigu_support
			}

			set_temp_variable = {
				var = temp_SOV_shoigu_influence
				value = SOV_shoigu_influence
			}

			add_to_temp_variable = {
				var = temp_SOV_shoigu_support
				value = temp_SOV_shoigu_influence
			}
			log = "total shoigu power: [?temp_SOV_shoigu_support]"

			set_variable = {
				var = SOV_shoigu_power
				value = temp_SOV_shoigu_support
			}

			set_temp_variable = {
				var = temp_SOV_gerasimov_support
				value = SOV_gerasimov_support
			}

			set_temp_variable = {
				var = temp_SOV_gerasimov_influence
				value = SOV_gerasimov_influence
			}

			add_to_temp_variable = {
				var = temp_SOV_gerasimov_support
				value = temp_SOV_gerasimov_influence
			}
			log = "total gerasimov power: [?temp_SOV_gerasimov_support]"

			set_variable = {
				var = SOV_gerasimov_power
				value = temp_SOV_gerasimov_support
			}

			if = {
				limit = {
					check_variable = {
            var = SOV_shoigu_power
            value = SOV_gerasimov_power
            compare = equals
           }
				}
				log = "no one dominates"
			}

			else_if = {
				log = "someone dominates"
			}
		}
	}
}
