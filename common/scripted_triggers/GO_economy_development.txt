# STATE. Type-specific snapshots extend, rather than replace, saved Services.
# Cancellation is possible without current GDP/levels/location/status.
GO_econ_development_project_type_record_valid = {
    OR = {
        check_variable = { GO_econ_state_project_type = 1 }
        AND = {
            check_variable = { GO_econ_state_project_type = 2 }
            has_variable = GO_econ_state_project_gdp_bonus
            check_variable = { var = GO_econ_state_project_gdp_bonus value = 0 compare = greater_than_or_equals }
        }
        AND = {
            check_variable = { GO_econ_state_project_type = 3 }
            has_variable = GO_econ_state_project_start_physical_level
            has_variable = GO_econ_state_project_expected_physical_next
            has_variable = GO_econ_state_project_building_grant_attempted
            check_variable = { GO_econ_state_project_next_level = GO_econ_state_project_expected_physical_next }
            OR = {
                AND = {
                    check_variable = { GO_econ_state_project_start_physical_level = 0 }
                    check_variable = { GO_econ_state_project_next_level = 1 }
                }
                AND = {
                    check_variable = { GO_econ_state_project_start_physical_level = 1 }
                    check_variable = { GO_econ_state_project_next_level = 2 }
                }
                AND = {
                    check_variable = { GO_econ_state_project_start_physical_level = 2 }
                    check_variable = { GO_econ_state_project_next_level = 3 }
                }
                AND = {
                    check_variable = { GO_econ_state_project_start_physical_level = 3 }
                    check_variable = { GO_econ_state_project_next_level = 4 }
                }
                AND = {
                    check_variable = { GO_econ_state_project_start_physical_level = 4 }
                    check_variable = { GO_econ_state_project_next_level = 5 }
                }
            }
            OR = {
                check_variable = { GO_econ_state_project_building_grant_attempted = 0 }
                check_variable = { GO_econ_state_project_building_grant_attempted = 1 }
            }
        }
    }
}
