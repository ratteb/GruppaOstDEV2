# COUNTRY internal enabled guard: caller MUST set both temporary candidate inputs.
# No has_variable probe is used to infer temporary-input existence.
GO_econ_ui_can_request_pu_allocation = {
    GO_econ_ui_pu_controls_ready = yes
    OR = {
        check_variable = { GO_econ_ui_candidate_kind = 1 }
        check_variable = { GO_econ_ui_candidate_kind = 2 }
    }
    check_variable = { var = GO_econ_ui_candidate_target value = 0 compare = greater_than_or_equals }
    check_variable = { var = GO_econ_ui_candidate_target value = GO_econ_country_pu_usable compare = less_than_or_equals }
    set_temp_variable = { GO_econ_ui_count_fraction = GO_econ_ui_candidate_target }
    modulo_temp_variable = { var = GO_econ_ui_count_fraction value = 1 }
    check_variable = { GO_econ_ui_count_fraction = 0 }
    set_temp_variable = { GO_econ_ui_candidate_civil_count = GO_econ_ui_candidate_target }
    if = {
        limit = { check_variable = { GO_econ_ui_candidate_kind = 2 } }
        set_temp_variable = { GO_econ_ui_candidate_civil_count = GO_econ_country_pu_usable }
        subtract_from_temp_variable = { var = GO_econ_ui_candidate_civil_count value = GO_econ_ui_candidate_target }
    }
    # ceil(100 * requested civil count / usable), then exact backend round trip.
    set_temp_variable = { GO_econ_ui_candidate_civil_percent = GO_econ_ui_candidate_civil_count }
    multiply_temp_variable = { var = GO_econ_ui_candidate_civil_percent value = 100 }
    divide_temp_variable = { var = GO_econ_ui_candidate_civil_percent value = GO_econ_country_pu_usable }
    set_temp_variable = { GO_econ_ui_count_fraction = GO_econ_ui_candidate_civil_percent }
    modulo_temp_variable = { var = GO_econ_ui_count_fraction value = 1 }
    subtract_from_temp_variable = { var = GO_econ_ui_candidate_civil_percent value = GO_econ_ui_count_fraction }
    if = {
        limit = { check_variable = { GO_econ_ui_count_fraction > 0 } }
        add_to_temp_variable = { var = GO_econ_ui_candidate_civil_percent value = 1 }
    }
    check_variable = { var = GO_econ_ui_candidate_civil_percent value = 0 compare = greater_than_or_equals }
    check_variable = { var = GO_econ_ui_candidate_civil_percent value = 100 compare = less_than_or_equals }
    set_temp_variable = { GO_econ_ui_count_round_trip = GO_econ_country_pu_usable }
    multiply_temp_variable = { var = GO_econ_ui_count_round_trip value = GO_econ_ui_candidate_civil_percent }
    divide_temp_variable = { var = GO_econ_ui_count_round_trip value = 100 }
    set_temp_variable = { GO_econ_ui_count_fraction = GO_econ_ui_count_round_trip }
    modulo_temp_variable = { var = GO_econ_ui_count_fraction value = 1 }
    subtract_from_temp_variable = { var = GO_econ_ui_count_round_trip value = GO_econ_ui_count_fraction }
    check_variable = { GO_econ_ui_count_round_trip = GO_econ_ui_candidate_civil_count }
}
