# COUNTRY. Stage 12 opt-in source. Policy 1 explicitly declares GDP-only income;
# policy 2 requires an additional authored current-serial income entry, even zero.
# Temporary arithmetic is observational. No account is credited by this guard.
GO_econ_country_gdp_monthly_income_ready = {
    GO_econ_country_gdp_income_source_ready = yes
    GO_econ_country_has_treasury = yes
    OR = {
        AND = {
            check_variable = { GO_econ_country_gdp_income_policy = 1 }
            # A populated/partial entry must be explicitly reconciled, not lost.
            has_variable = GO_econ_country_monthly_receipts_populated
            has_variable = GO_econ_country_receipts_month_populated
            check_variable = { GO_econ_country_monthly_receipts_populated = 0 }
            check_variable = { GO_econ_country_receipts_month_populated = 0 }
        }
        AND = {
            check_variable = { GO_econ_country_gdp_income_policy = 2 }
            GO_econ_country_authored_monthly_income_ready = yes
        }
    }
    # Do not enter division/multiplication with an invalid or missing basis.
    if = {
        limit = {
            GO_econ_country_has_gdp = yes
            GO_econ_country_has_price_index = yes
            has_variable = global.GO_econ_cfg_gdp_income_factor
            check_variable = { var = global.GO_econ_cfg_gdp_income_factor value = 0 compare = greater_than_or_equals }
            has_variable = global.GO_econ_cfg_months_per_year
            check_variable = { global.GO_econ_cfg_months_per_year > 0 }
        }
        set_temp_variable = { GO_econ_calc_gdp_monthly_income = GO_econ_country_gdp }
        multiply_temp_variable = { GO_econ_calc_gdp_monthly_income = GO_econ_country_price_index }
        multiply_temp_variable = { GO_econ_calc_gdp_monthly_income = global.GO_econ_cfg_gdp_income_factor }
        divide_temp_variable = { GO_econ_calc_gdp_monthly_income = global.GO_econ_cfg_months_per_year }
        check_variable = { var = GO_econ_calc_gdp_monthly_income value = 0 compare = greater_than_or_equals }
        if = {
            limit = {
                check_variable = { GO_econ_country_gdp > 0 }
                check_variable = { global.GO_econ_cfg_gdp_income_factor > 0 }
            }
            check_variable = { GO_econ_calc_gdp_monthly_income > 0 }
        }
        set_temp_variable = { GO_econ_calc_combined_monthly_income = GO_econ_calc_gdp_monthly_income }
        if = {
            limit = { check_variable = { GO_econ_country_gdp_income_policy = 2 } }
            add_to_temp_variable = { GO_econ_calc_combined_monthly_income = GO_econ_country_monthly_receipts }
            check_variable = { var = GO_econ_calc_combined_monthly_income value = GO_econ_country_monthly_receipts compare = greater_than_or_equals }
            if = {
                limit = { check_variable = { GO_econ_country_monthly_receipts > 0 } }
                check_variable = { GO_econ_calc_combined_monthly_income > GO_econ_calc_gdp_monthly_income }
            }
        }
        check_variable = { var = GO_econ_calc_combined_monthly_income value = GO_econ_calc_gdp_monthly_income compare = greater_than_or_equals }
        set_temp_variable = { GO_econ_calc_income_treasury = GO_econ_country_treasury }
        add_to_temp_variable = { GO_econ_calc_income_treasury = GO_econ_calc_combined_monthly_income }
        check_variable = { var = GO_econ_calc_income_treasury value = GO_econ_country_treasury compare = greater_than_or_equals }
        check_variable = { var = GO_econ_calc_income_treasury value = GO_econ_calc_combined_monthly_income compare = greater_than_or_equals }
        if = {
            limit = { check_variable = { GO_econ_calc_combined_monthly_income > 0 } }
            check_variable = { GO_econ_calc_income_treasury > GO_econ_country_treasury }
        }
    }
}
