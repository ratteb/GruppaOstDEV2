# Gruppa Ost Stage 6 - COUNTRY national material account guards.
# Abstract game-accounting units, not mass. Fractional nonnegative balances and
# source quantities are permitted; assigned ammunition levels use whole guards
# in their separate backend. No state resource_* values enter these accounts.
# Established variable/marker syntax: design evidence E-001/E-002/E-009.
# Temporary arithmetic follows installed 1.18.3.0.7709 / E-028.
#
# Catalog IDs: 1 iron, 2 bauxite, 3 food, 4 coal, 5 natural_gas, 6 oil,
# 7 rare_earth_minerals, 8 tungsten, 9 titanium, 10 steel, 11 aluminium,
# 12 rubber, 13 rations, 14 fuel, 15 microchips.
# A complete ready account includes stock plus lifetime flow history since its
# explicit opening. Counters are never a daily throughput or a complete budget.

GO_econ_country_materials_initialized = {
    has_variable = GO_econ_country_material_schema_version
    check_variable = { GO_econ_country_material_schema_version = 1 }
}

# Stage 7 - explicitly authored daily supply and coordinated processing.
# COUNTRY only; regular selectors retain the Stage 6 catalog.
# Supply attempts and successful receipts have separate persistent dates.
# Positive transactions/deltas must change stock and history in their actual
# direction; a zero quantity may leave both unchanged. No numeric cap is guessed.
# Recipe account preflight includes subtraction arithmetic before distinguishing
# sufficient stock from a known shortage. These checks do not prove native bounds.
# Native-day and prospective temporary arithmetic follow E-021/E-028/E-033.

GO_econ_material_native_day_ready = {
    check_variable = { var = global.num_days value = 0 compare = greater_than_or_equals }
    set_temp_variable = { GO_econ_material_native_day_fraction = global.num_days }
    modulo_temp_variable = { var = GO_econ_material_native_day_fraction value = 1 }
    check_variable = { GO_econ_material_native_day_fraction = 0 }
}
