    has_variable = global.GO_econ_month_serial
    has_variable = global.GO_econ_month_pulse_day
    has_variable = global.GO_econ_daily_pulse_day
    check_variable = { var = global.GO_econ_month_serial value = 0 compare = greater_than_or_equals }
    check_variable = { var = global.GO_econ_month_pulse_day value = -1 compare = greater_than_or_equals }
    check_variable = { var = global.GO_econ_month_pulse_day value = global.num_days compare = less_than_or_equals }
    check_variable = { var = global.GO_econ_daily_pulse_day value = -1 compare = greater_than_or_equals }
    check_variable = { var = global.GO_econ_daily_pulse_day value = global.num_days compare = less_than_or_equals }
}
