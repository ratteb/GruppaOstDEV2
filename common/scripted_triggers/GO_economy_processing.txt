# Shift/clamp only temporary -1 day copies before modulo, following E-043.
# Native state building levels and day identity follow E-028/E-029/E-034.
# Missing or fractional physical levels/assignments are unavailable, never zero.
# Historical reports validate their snapshots, independent of current tuning.

# STATE. Physical zero skips only an absent or genuinely unused record.
# Invalid/partial saved records cannot disguise themselves as unassigned zero.
GO_econ_state_steel_zero_capacity_unassigned = {
    check_variable = { building_level@GO_econ_steel_plant = 0 }
    OR = {
        AND = {
            NOT = { has_variable = GO_econ_state_steel_assignment_schema_version }
            NOT = { has_variable = GO_econ_state_steel_assigned_levels }
            NOT = { has_variable = GO_econ_state_steel_assignment_populated }
            NOT = { has_variable = GO_econ_state_steel_assignment_owner }
        }
        AND = {
            GO_econ_state_steel_assignment_initialized = yes
            has_variable = GO_econ_state_steel_assigned_levels
            has_variable = GO_econ_state_steel_assignment_populated
            check_variable = { GO_econ_state_steel_assigned_levels = 0 }
            check_variable = { GO_econ_state_steel_assignment_populated = 0 }
            NOT = { has_variable = GO_econ_state_steel_assignment_owner }
        }
    }
}
