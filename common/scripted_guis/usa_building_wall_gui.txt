scripted_gui = {


	USA_building_wall_gui = {
    window_name = "USA_building_wall_gui_window"
    context_type = decision_category

    visible = {
      has_country_flag = USA_building_wall
    }

    effects = {

      increase_button_click = {
			subtract_from_variable = {
        var = USA_build_wall_funds
        value = 500
      }
      add_to_variable = {
        var = USA_build_wall_investment_points
        value = 1 
      }
      log = "USA bought IP" 
		}

    decrease_button_click = {
			add_to_variable = {
        var = USA_build_wall_funds
        value = 500
      }
      subtract_from_variable = {
        var = USA_build_wall_investment_points
        value = 1 
      }
      log = "USA sold IP" 
		}
    }

    triggers = {
      increase_button_click_enabled = {
        check_variable = {
          var = USA_build_wall_funds 
          value = 500
          compare = greater_than_or_equals
        }
        NOT = {
          check_variable = {
            var = USA_build_wall_investment_points
            value = 10 
            compare = greater_than_or_equals 
          }
        }
      }

      decrease_button_click_enabled = {
        
          check_variable = {
            var = USA_build_wall_investment_points
            value = 1 
            compare = lesser_than_or_equals 
          }
        
      }
    }


    dynamic_lists = {
	  }

    properties = {  
      usa_wall_progress_0_25_meter = {
        frame = USA_build_wall_progress
        }
	}

    ai_enabled = {
      always = yes 
    }

    ai_test_interval = 24  # The higher, the long it takes the ai to do actions
	ai_test_variance = 0.5

    ai_check = {
    }

    ai_check_scope = {
    }

    ai_weights = {
      increase_button_click = {
      ai_will_do = {
			base = 10
			modifer = {
				check_variable = {
					var = USA_build_wall_investment_points
					value = 10
					compare = equals
				}
				add = -10
			}
		}
      }

      decrease_button_click_button_click = {
      ai_will_do = {
			base = 0
			modifer = {
				check_variable = {
					var = USA_build_wall_investment_points
					value = 10
					compare = equals
				}
				add = 10
			}
		}
      }
    }
  }
}
