# Gruppa Ost Stage 6 - national steel-funded physical ammunition production.
# The sole ammunition stock remains ammunition_variable, measured in thousands.
# Plant output joins ammunition_daily AFTER legacy gain modifiers, before its
# one stock credit and existing capacity clamp. Gross production may exceed room.
# Sources: installed 1.18.3.0.7709 dynamic_variables_documentation.md:990-991,
# 1095-1096; effects_documentation.md:4046-4064 (owned-state scope).

# STATE. Storage is unavailable until an authored assignment, including zero.
# Existing assignments/schema/owner identities are preserved, not guessed.
GO_econ_initialize_state_ammunition_assignment = {
    if = {
        limit = {
            NOT = { has_variable = GO_econ_state_ammunition_assignment_schema_version }
            NOT = { has_variable = GO_econ_state_ammunition_assigned_levels }
            NOT = { has_variable = GO_econ_state_ammunition_assignment_populated }
            NOT = { has_variable = GO_econ_state_ammunition_assignment_owner }
        }
        set_variable = { GO_econ_state_ammunition_assigned_levels = 0 }
        set_variable = { GO_econ_state_ammunition_assignment_populated = 0 }
        set_variable = { GO_econ_state_ammunition_assignment_schema_version = 1 }
    }
}

# COUNTRY. Fresh-only replay metadata; no ammunition/material stock is created.
GO_econ_initialize_country_ammunition_production = {
    if = {
        limit = {
            NOT = { has_variable = GO_econ_country_ammunition_schema_version }
            NOT = { has_variable = GO_econ_country_ammunition_last_day }
            NOT = { has_variable = GO_econ_country_ammunition_plant_last_day }
            NOT = { has_variable = GO_econ_country_ammunition_plant_report_day }
            NOT = { has_variable = GO_econ_country_ammunition_plant_report_populated }
            NOT = { has_variable = GO_econ_country_ammunition_plant_report_built }
            NOT = { has_variable = GO_econ_country_ammunition_plant_report_assigned }
            NOT = { has_variable = GO_econ_country_ammunition_plant_report_capacity_populated }
            NOT = { has_variable = GO_econ_country_ammunition_plant_operating }
            NOT = { has_variable = GO_econ_country_ammunition_plant_steel_consumed }
            NOT = { has_variable = GO_econ_country_ammunition_plant_output_k }
            NOT = { has_variable = GO_econ_country_ammunition_plant_status }
            NOT = { has_variable = GO_econ_country_ammunition_plant_capacity_populated }
            NOT = { has_variable = GO_econ_country_ammunition_plant_eligible_built }
            NOT = { has_variable = GO_econ_country_ammunition_plant_assigned }
            NOT = { has_variable = GO_econ_country_ammunition_plant_missing_assignments }
        }
        set_variable = { GO_econ_country_ammunition_schema_version = 1 }
        set_variable = { GO_econ_country_ammunition_last_day = -1 }
        set_variable = { GO_econ_country_ammunition_plant_last_day = -1 }
        set_variable = { GO_econ_country_ammunition_plant_report_day = -1 }
        set_variable = { GO_econ_country_ammunition_plant_report_populated = 0 }
        set_variable = { GO_econ_country_ammunition_plant_report_capacity_populated = 0 }
        set_variable = { GO_econ_country_ammunition_plant_report_built = 0 }
        set_variable = { GO_econ_country_ammunition_plant_report_assigned = 0 }
    }
    if = {
        limit = { NOT = { has_variable = GO_econ_country_ammunition_plant_capacity_populated } }
        set_variable = { GO_econ_country_ammunition_plant_capacity_populated = 0 }
    }
}

# COUNTRY anchor/ANY nested scopes; called from the formerly empty ammo startup.
# Only unavailable ledger/assignment storage and fresh replay metadata are added.
GO_econ_initialize_ammunition_backend = {
    every_state = { GO_econ_initialize_state_ammunition_assignment = yes }
    every_country = {
        GO_econ_initialize_country_materials = yes
        GO_econ_initialize_country_ammunition_production = yes
    }
}

# STATE public API. Caller sets actor country and whole assigned level count.
# Successful zero explicitly records unassigned capacity; no material is spent.
GO_econ_assign_ammunition_plant = {
    set_variable = { GO_econ_work_ammunition_assignment_succeeded = 0 }
    GO_econ_initialize_state_ammunition_assignment = yes
    if = {
        limit = { GO_econ_can_assign_ammunition_plant = yes }
        set_variable = { GO_econ_state_ammunition_assignment_owner = GO_econ_work_actor }
        set_variable = { GO_econ_state_ammunition_assigned_levels = GO_econ_work_ammunition_assigned_levels }
        set_variable = { GO_econ_state_ammunition_assignment_populated = 1 }
        set_variable = { GO_econ_work_ammunition_assignment_succeeded = 1 }
        var:GO_econ_work_actor = {
            GO_econ_initialize_country_ammunition_production = yes
            GO_econ_refresh_ammunition_plant_capacity = yes
        }
    }
}

# COUNTRY. Rebuild current owned-and-controlled capacity without producing.
# Incomplete assignments on physical plants block the entire national batch.
# Captured states retain assignments; transferred states need authored reassignment.
GO_econ_refresh_ammunition_plant_capacity = {
    GO_econ_initialize_country_ammunition_production = yes
    set_variable = { GO_econ_country_ammunition_plant_capacity_populated = 0 }
    set_variable = { GO_econ_country_ammunition_plant_eligible_built = 0 }
    set_variable = { GO_econ_country_ammunition_plant_assigned = 0 }
    set_variable = { GO_econ_country_ammunition_plant_missing_assignments = 0 }
    if = {
        limit = { GO_econ_country_ammunition_production_initialized = yes }
        every_owned_state = {
            limit = { is_controlled_by = PREV }
            if = {
                limit = { GO_econ_state_ammunition_physical_levels_valid = yes }
                add_to_variable = { PREV.GO_econ_country_ammunition_plant_eligible_built = building_level@ammunition_plant }
                if = {
                    limit = {
                        OR = {
                            check_variable = { building_level@ammunition_plant > 0 }
                            NOT = { GO_econ_state_ammunition_has_no_assignment = yes }
                        }
                    }
                    if = {
                        limit = { GO_econ_state_has_ammunition_assignment = yes }
                        add_to_variable = { PREV.GO_econ_country_ammunition_plant_assigned = GO_econ_state_ammunition_assigned_levels }
                    }
                    else = {
                        add_to_variable = { PREV.GO_econ_country_ammunition_plant_missing_assignments = 1 }
                    }
                }
            }
            else = {
                add_to_variable = { PREV.GO_econ_country_ammunition_plant_missing_assignments = 1 }
            }
        }
        if = {
            limit = { check_variable = { GO_econ_country_ammunition_plant_missing_assignments = 0 } }
            set_variable = { GO_econ_country_ammunition_plant_capacity_populated = 1 }
            if = {
                limit = { NOT = { GO_econ_country_ammunition_plant_capacity_ready = yes } }
                set_variable = { GO_econ_country_ammunition_plant_capacity_populated = 0 }
            }
        }
    }
}

# COUNTRY internal daily stage. The legacy hook has already claimed its day and
# calculated its modifier-based gain. Claim recipe day before input inspection;
# blocked attempts take effect next day after repair, never as a second deposit.
GO_econ_update_ammunition_plant_production = {
    if = {
        limit = { GO_econ_ammunition_plant_daily_update_ready = yes }
        set_variable = { GO_econ_country_ammunition_plant_last_day = global.num_days }
        set_variable = { GO_econ_country_ammunition_plant_report_day = global.num_days }
        set_variable = { GO_econ_country_ammunition_plant_report_populated = 0 }
        set_variable = { GO_econ_country_ammunition_plant_report_capacity_populated = 0 }
        set_variable = { GO_econ_country_ammunition_plant_report_built = 0 }
        set_variable = { GO_econ_country_ammunition_plant_report_assigned = 0 }
        set_variable = { GO_econ_country_ammunition_plant_operating = 0 }
        set_variable = { GO_econ_country_ammunition_plant_steel_consumed = 0 }
        set_variable = { GO_econ_country_ammunition_plant_output_k = 0 }
        set_variable = { GO_econ_country_ammunition_plant_status = 0 }
        GO_econ_refresh_ammunition_plant_capacity = yes
        if = {
            limit = { GO_econ_country_ammunition_plant_capacity_ready = yes }
            set_variable = { GO_econ_country_ammunition_plant_report_built = GO_econ_country_ammunition_plant_eligible_built }
            set_variable = { GO_econ_country_ammunition_plant_report_assigned = GO_econ_country_ammunition_plant_assigned }
            set_variable = { GO_econ_country_ammunition_plant_report_capacity_populated = 1 }
        }
        if = {
            limit = {
                GO_econ_country_ammunition_plant_capacity_ready = yes
                GO_econ_country_has_material_steel = yes
            }
            if = {
                limit = {
                    check_variable = { var = GO_econ_country_material_steel value = GO_econ_country_ammunition_plant_assigned compare = greater_than_or_equals }
                }
                if = {
                    limit = { GO_econ_ammunition_plant_recipe_numbers_ready = yes }
                    # Full family batch only. The Stage 7 dispatcher places
                    # this steel consumer before microchips, without rationing.
                    set_variable = { GO_econ_work_material_id = 10 }
                    set_variable = { GO_econ_work_material_quantity = GO_econ_country_ammunition_plant_assigned }
                    GO_econ_consume_national_material = yes
                    if = {
                        limit = { check_variable = { GO_econ_work_material_succeeded = 1 } }
                        set_variable = { GO_econ_country_ammunition_plant_operating = GO_econ_country_ammunition_plant_assigned }
                        set_variable = { GO_econ_country_ammunition_plant_steel_consumed = GO_econ_country_ammunition_plant_operating }
                        set_variable = { GO_econ_country_ammunition_plant_output_k = GO_econ_country_ammunition_plant_operating }
                        multiply_variable = { GO_econ_country_ammunition_plant_output_k = 0.5 }
                        add_to_variable = { ammunition_daily = GO_econ_country_ammunition_plant_output_k }
                        set_variable = { GO_econ_country_ammunition_plant_status = 1 }
                        set_variable = { GO_econ_country_ammunition_plant_report_populated = 1 }
                    }
                }
            }
            else = {
                # Known shortage yields known zero use/output; unavailable
                # steel/assignment instead retains report_populated = 0 above.
                set_variable = { GO_econ_country_ammunition_plant_status = 2 }
                set_variable = { GO_econ_country_ammunition_plant_report_populated = 1 }
            }
        }
    }
}

# COUNTRY. Stage 7 coordinator calls the preserved legacy daily ammunition path.
# Leader counters use PREV (the actual country) so nested callers are safe.
# Fixed plant output remains after the legacy multiplier and before one stock credit.
GO_econ_update_daily_ammunition = {
	GO_econ_initialize_country_ammunition_production = yes
	if = {
		limit = { GO_econ_ammunition_daily_update_ready = yes }
		# Persist before consumption/receipts; repeated calls cannot settle twice.
		set_variable = { GO_econ_country_ammunition_last_day = global.num_days }


	#### daily consumption

	set_variable = { ammunition_consumption = 0 }

	set_variable = { infantry_consumption = num_battalions_with_type@infantry }
	multiply_variable = { infantry_consumption = 0.01 }
	set_variable = { artillery_consumption = num_battalions_with_type@artillery }
	multiply_variable = { artillery_consumption = 0.1 }
	set_variable = { light_consumption = num_battalions_with_type@light_armor }
	multiply_variable = { light_consumption = 0.04 }
	set_variable = { medium_consumption = num_battalions_with_type@medium_armor }
	multiply_variable = { medium_consumption = 0.06 }
	set_variable = { heavy_consumption = num_battalions_with_type@heavy_armor }
	multiply_variable = { heavy_consumption = 0.08 }
	set_variable = { remaining_consumption = num_battalions }
	subtract_from_variable = { remaining_consumption = num_battalions_with_type@infantry }
	subtract_from_variable = { remaining_consumption = num_battalions_with_type@artillery }
	subtract_from_variable = { remaining_consumption = num_battalions_with_type@light_armor }
	subtract_from_variable = { remaining_consumption = num_battalions_with_type@medium_armor }
	subtract_from_variable = { remaining_consumption = num_battalions_with_type@heavy_armor }
	multiply_variable = { remaining_consumption = 0.02 }

	set_variable = { potential_consumption = 0 }
	add_to_variable = { potential_consumption = infantry_consumption }
	add_to_variable = { potential_consumption = artillery_consumption }
	add_to_variable = { potential_consumption = light_consumption }
	add_to_variable = { potential_consumption = medium_consumption }
	add_to_variable = { potential_consumption = heavy_consumption }
	add_to_variable = { potential_consumption = remaining_consumption }
	set_variable = { potential_consumption_max = potential_consumption }
	multiply_variable = { potential_consumption_max = 1.5 }
	divide_variable = { potential_consumption = num_divisions }

	set_variable = { commanded_divisions = 0 }
	set_variable = { divisions_in_offensive_combat = 0 }
	set_variable = { divisions_in_defensive_combat = 0 }
	every_army_leader = {
		limit = {
			is_leading_army_group = no
		}
		add_to_variable = { PREV.commanded_divisions = num_units }
		add_to_variable = { PREV.divisions_in_offensive_combat = num_units_offensive_combats }
		add_to_variable = { PREV.divisions_in_defensive_combat = num_units_defensive_combats }
	}


	#### leaderless consumption
	set_variable = { divisions_without_leadership = num_divisions }
	subtract_from_variable = { divisions_without_leadership = commanded_divisions }
	multiply_variable = { divisions_without_leadership = potential_consumption }
	multiply_variable = { divisions_without_leadership = 1 }

	#### offensive units consumption
	multiply_variable = { divisions_in_offensive_combat = potential_consumption }
	multiply_variable = { divisions_in_offensive_combat = 1.5 }

	#### offensive units consumption
	multiply_variable = { divisions_in_defensive_combat = potential_consumption }
	multiply_variable = { divisions_in_defensive_combat = 0.75 }

	set_variable = { daily_ammo_need = divisions_without_leadership }
	add_to_variable = { daily_ammo_need = divisions_in_offensive_combat }
	add_to_variable = { daily_ammo_need = divisions_in_defensive_combat }
	clamp_variable = {
		var = daily_ammo_need
		min = 0
	}


	subtract_from_variable = { ammunition_variable = daily_ammo_need }


	################## end daily consumption




	#### daily addition

	set_variable = { ammunition_daily = 0 }

	# base gain
	set_variable = { modifier_gain = modifier@daily_ammunition_gain }
	divide_variable = { modifier_gain = 1000 }
	add_to_variable = { ammunition_daily = modifier_gain }

	# factor
	set_variable = { ammunition_factor_calc = 1 }
	add_to_variable = { ammunition_factor_calc = modifier@ammunition_factor }
	clamp_variable = {
		var = ammunition_factor_calc
		min = 0
		#max = 0
	}

	multiply_variable = { ammunition_daily = ammunition_factor_calc }

	# Fixed 500/day plant output is outside the legacy gain factor.
	# The recipe debits steel once and joins the single stock credit below.
	GO_econ_update_ammunition_plant_production = yes

	add_to_variable = { ammunition_variable = ammunition_daily }

	################## end daily addition



	set_variable = { ammunition_balance = ammunition_daily }
	subtract_from_variable = { ammunition_balance = daily_ammo_need }
	set_variable = { ammunition_balance_int = ammunition_balance }
	if = {
		limit = {
			check_variable = { ammunition_balance < 0 }
		}
		multiply_variable = { ammunition_balance_int = -1 }
	}



	set_variable = { ammunition_max = 0 }
	add_to_variable = { ammunition_max = owned_controlled_states }
	multiply_variable = { ammunition_max = 10 }
	set_variable = { ammunition_storage = modifier@ammunition_capacity_k }
	add_to_variable = { ammunition_max = ammunition_storage }

	clamp_variable = {
		var = ammunition_variable
		min = 0
		max = ammunition_max
	}




	set_variable = { ammunition_penalty = daily_ammo_need }
	subtract_from_variable = { ammunition_penalty = ammunition_daily }
	divide_variable = { ammunition_penalty = daily_ammo_need }
	multiply_variable = { ammunition_penalty = -1 }
	if = {
		limit = {
			check_variable = { ammunition_balance < 0 }
			check_variable = { ammunition_variable < ammunition_balance_int }
			NOT = {
				has_dynamic_modifier = { modifier = ammunition_effects }
			}
		}
		add_dynamic_modifier = { modifier = ammunition_effects }
	}


	##### interface

	set_variable = { ammo_h = ammunition_variable }
	multiply_variable = { ammo_h = 1000 }

	set_variable = { ammo_daily_h = ammunition_daily }
	multiply_variable = { ammo_daily_h = 1000 }

	set_variable = { ammo_need_h = daily_ammo_need }
	multiply_variable = { ammo_need_h = 1000 }

	set_variable = { potential_need_h = potential_consumption_max }
	multiply_variable = { potential_need_h = 1000 }

	set_variable = { ammunition_max_h = ammunition_max }
	multiply_variable = { ammunition_max_h = 1000 }

	set_variable = { potential_consumption_h = potential_consumption_max }
	multiply_variable = { potential_consumption_h = 1000 }

	set_variable = { ammo_ratio_var = ammunition_variable }
	divide_variable = { ammo_ratio_var = ammunition_max }
	multiply_variable = { ammo_ratio_var = 100 }

	set_variable = { ammo_days_to_full = ammunition_max }
	subtract_from_variable = { ammo_days_to_full = ammunition_variable }
	divide_variable = { ammo_days_to_full = ammunition_balance }

	set_variable = { ammo_days_to_empty = ammunition_variable }
	divide_variable = { ammo_days_to_empty = ammunition_balance }
	multiply_variable = { ammo_days_to_empty = -1 }

	if = {
		limit = {
			check_variable = { ammunition_balance > 0 }
		}
		set_variable = { ammo_frame_var = 2 }
	}
	if = {
		limit = {
			check_variable = { ammunition_variable = ammunition_max }
		}
		set_variable = { ammo_frame_var = 1 }
	}
	if = {
		limit = {
			check_variable = { ammunition_balance < 0 }
		}
		set_variable = { ammo_frame_var = 3 }
	}
	}
}
