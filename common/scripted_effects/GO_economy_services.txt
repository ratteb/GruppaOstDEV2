# Gruppa Ost - Government Services state project backend, HOI4 1.18.3.
# No decisions, UI, initial service levels, tax/receipt data or population dataset.
# Public start/cancel: STATE scope; caller sets regular GO_econ_work_actor to its
# country target. Paid record remains historical when active/type become zero.
# Project fields use GO_econ_state_project_: active/type/sponsor/paid_cost,
# cash_used/borrowed_outstanding/required_days/workdays/last_work_day/next_level,
# start_gdp/start_price/completed/refund_pending/completion_pending.
# Start payment helper GO_econ_pay_state_cost accepts actor + GO_econ_work_cost;
# returns GO_econ_work_payment_ok/cash_used/borrowed without booking flow categories.
# State liabilities are canonical GO_econ_state_liability_@debtor; country debt
# and state geographic debt are rebuildable reports, never independent balances.
# Sources: installed effects_documentation.md:2674-2688,4125-4140;
# USA_scripted_effects.txt:71 (variable clamp), NORDIC_scripted_effects.txt:473,
# 508-514 (stored country scopes), WTT_china_political_struggle_effects.txt:80-86
# (country-derived state keys), dynamic_variables_documentation.md:990-991.

# STATE. Integer checks are derived from the live saved record at every caller.
# These remainders are validation caches, not authored economic inputs. Installed
# effects_documentation.md:4969-4983 documents modulo_variable.
GO_econ_refresh_services_project_record = {
    if = {
        limit = {
            GO_econ_state_fiscal_initialized = yes
            check_variable = { GO_econ_state_project_active = 1 }
            has_variable = GO_econ_state_project_required_days
            has_variable = GO_econ_state_project_workdays
            has_variable = GO_econ_state_project_last_work_day
            has_variable = GO_econ_state_project_cash_used
            has_variable = GO_econ_state_project_borrowed_outstanding
        }
        set_variable = { GO_econ_state_project_required_days_fraction = GO_econ_state_project_required_days }
        modulo_variable = { GO_econ_state_project_required_days_fraction = 1 }
        set_variable = { GO_econ_state_project_workdays_fraction = GO_econ_state_project_workdays }
        modulo_variable = { GO_econ_state_project_workdays_fraction = 1 }
        set_variable = { GO_econ_state_project_last_work_day_fraction = GO_econ_state_project_last_work_day }
        modulo_variable = { GO_econ_state_project_last_work_day_fraction = 1 }
        set_variable = { GO_econ_state_project_funding_total = GO_econ_state_project_cash_used }
        add_to_variable = { GO_econ_state_project_funding_total = GO_econ_state_project_borrowed_outstanding }
        if = {
            limit = {
                check_variable = { GO_econ_state_project_type = 3 }
                has_variable = GO_econ_state_project_start_physical_level
            }
            set_variable = { GO_econ_state_project_expected_physical_next = GO_econ_state_project_start_physical_level }
            add_to_variable = { GO_econ_state_project_expected_physical_next = 1 }
        }
    }
}

# COUNTRY. Rebuild all sponsored active projects, including paused/lost states.
# An unclassifiable live record blocks capacity rather than silently freeing slots.
GO_econ_refresh_project_slots = {
    set_variable = { GO_econ_country_active_projects = 0 }
    set_variable = { GO_econ_country_active_projects_populated = 1 }
    every_state = {
        GO_econ_refresh_services_project_record = yes
        if = {
            limit = {
                OR = {
                    NOT = { has_variable = GO_econ_state_project_active }
                    AND = {
                        NOT = { check_variable = { GO_econ_state_project_active = 0 } }
                        NOT = { check_variable = { GO_econ_state_project_active = 1 } }
                    }
                    AND = {
                        check_variable = { GO_econ_state_project_active = 1 }
                        NOT = { GO_econ_services_project_has_sponsor = yes }
                    }
                    AND = {
                        check_variable = { GO_econ_state_project_active = 0 }
                        OR = {
                            NOT = { has_variable = GO_econ_state_project_type }
                            NOT = { check_variable = { GO_econ_state_project_type = 0 } }
                        }
                    }
                }
            }
            set_variable = { PREV.GO_econ_country_active_projects_populated = 0 }
        }
        else_if = {
            limit = {
                check_variable = { GO_econ_state_project_active = 1 }
                check_variable = { GO_econ_state_project_sponsor = PREV }
            }
            if = {
                limit = { GO_econ_development_project_record_valid = yes }
                add_to_variable = { PREV.GO_econ_country_active_projects = 1 }
            }
            else = {
                set_variable = { PREV.GO_econ_country_active_projects_populated = 0 }
            }
        }
    }
}

# STATE. Recheck inside the effect and reserve before making the one payment.
GO_econ_start_services_project = {
    GO_econ_initialize_fiscal_config = yes
    GO_econ_initialize_state_fiscal = yes
    GO_econ_reconcile_services_owner = yes
    GO_econ_refresh_native_population = yes
    GO_econ_refresh_state_debt = yes
    if = {
        limit = { GO_econ_work_actor_is_country = yes }
        var:GO_econ_work_actor = {
            GO_econ_initialize_country_fiscal = yes
            GO_econ_refresh_country_debt = yes
            GO_econ_refresh_status_limits = yes
            GO_econ_refresh_project_slots = yes
        }
    }
    GO_econ_refresh_state_gdp = yes
    if = {
        limit = { GO_econ_can_start_services_project = yes }
        set_variable = { GO_econ_state_project_active = 1 }
        set_variable = { GO_econ_state_project_type = 1 }
        set_variable = { GO_econ_state_project_sponsor = GO_econ_work_actor }
        set_variable = { GO_econ_state_project_completed = 0 }
        set_variable = { GO_econ_state_project_refund_pending = 0 }
        set_variable = { GO_econ_state_project_completion_pending = 0 }
        set_variable = { GO_econ_state_project_next_level = GO_econ_state_services_completed_levels }
        add_to_variable = { GO_econ_state_project_next_level = 1 }
        set_variable = { GO_econ_state_project_start_gdp = GO_econ_state_gdp }
        set_variable = { GO_econ_state_project_workdays = 0 }
        set_variable = { GO_econ_state_project_required_days = global.GO_econ_cfg_services_required_days }
        set_variable = { GO_econ_state_project_last_work_day = global.num_days }
        var:GO_econ_work_actor = {
            PREV = {
                set_variable = { GO_econ_state_project_start_price = PREV.GO_econ_country_price_index }
            }
        }
        set_variable = { GO_econ_work_cost = GO_econ_state_project_start_gdp }
        multiply_variable = { GO_econ_work_cost = global.GO_econ_cfg_services_cost_factor }
        multiply_variable = { GO_econ_work_cost = GO_econ_state_project_next_level }
        clamp_variable = { var = GO_econ_work_cost min = global.GO_econ_cfg_services_min_cost }
        multiply_variable = { GO_econ_work_cost = GO_econ_state_project_start_price }
        set_variable = { GO_econ_state_project_paid_cost = GO_econ_work_cost }
        set_variable = { GO_econ_state_project_cash_used = 0 }
        set_variable = { GO_econ_state_project_borrowed_outstanding = 0 }
        GO_econ_refresh_services_project_record = yes
        set_variable = { GO_econ_work_payment_ok = 0 }
        GO_econ_pay_state_cost = yes
        if = {
            limit = { check_variable = { GO_econ_work_payment_ok = 1 } }
            set_variable = { GO_econ_state_project_cash_used = GO_econ_work_cash_used }
            set_variable = { GO_econ_state_project_borrowed_outstanding = GO_econ_work_borrowed }
            var:GO_econ_state_project_sponsor = {
                GO_econ_mark_open_fiscal_flow = yes
                add_to_variable = { GO_econ_country_capital_expenses_pending = PREV.GO_econ_state_project_paid_cost }
                add_to_variable = { GO_econ_country_cash_used_pending = PREV.GO_econ_state_project_cash_used }
                add_to_variable = { GO_econ_country_borrowing_pending = PREV.GO_econ_state_project_borrowed_outstanding }
                GO_econ_refresh_project_slots = yes
                GO_econ_refresh_fiscal_report = yes
            }
        }
        else = {
            # The payment helper guarantees no money changes when it rejects.
            set_variable = { GO_econ_state_project_active = 0 }
            set_variable = { GO_econ_state_project_type = 0 }
            var:GO_econ_state_project_sponsor = { GO_econ_refresh_project_slots = yes }
        }
    }
}

# STATE, internal. Actor identity/finance are checked again before any refund.
# This helper accepts lost legal ownership and enemy control by design.
GO_econ_refund_development_project = {
    GO_econ_refresh_services_project_record = yes
    GO_econ_refresh_state_debt = yes
    if = {
        limit = { GO_econ_services_project_has_sponsor = yes }
        var:GO_econ_state_project_sponsor = { GO_econ_refresh_country_debt = yes }
    }
    if = {
        limit = { GO_econ_can_cancel_development_project = yes }
        # Clear the live lock before benefits: repeated calls cannot refund again.
        set_variable = { GO_econ_state_project_active = 0 }
        set_variable = { GO_econ_state_project_type = 0 }
        set_variable = { GO_econ_state_project_refund_pending = 0 }
        set_variable = { GO_econ_state_project_completion_pending = 0 }
        set_variable = { GO_econ_work_refund = GO_econ_state_project_workdays }
        divide_variable = { GO_econ_work_refund = GO_econ_state_project_required_days }
        multiply_variable = { GO_econ_work_refund = -1 }
        add_to_variable = { GO_econ_work_refund = 1 }
        multiply_variable = { GO_econ_work_refund = GO_econ_state_project_paid_cost }
        clamp_variable = { var = GO_econ_work_refund min = 0 max = GO_econ_state_project_paid_cost }
        set_variable = { GO_econ_work_debt_retired = GO_econ_work_refund }
        clamp_variable = { var = GO_econ_work_debt_retired min = 0 max = GO_econ_state_project_borrowed_outstanding }
        var:GO_econ_state_project_sponsor = {
            GO_econ_mark_open_fiscal_flow = yes
            PREV = {
                clamp_variable = { var = GO_econ_work_debt_retired min = 0 max = GO_econ_state_liability_@PREV }
                subtract_from_variable = { GO_econ_state_liability_@PREV = GO_econ_work_debt_retired }
                subtract_from_variable = { GO_econ_state_project_borrowed_outstanding = GO_econ_work_debt_retired }
                set_variable = { GO_econ_work_refund_cash = GO_econ_work_refund }
                subtract_from_variable = { GO_econ_work_refund_cash = GO_econ_work_debt_retired }
                add_to_variable = { PREV.GO_econ_country_treasury = GO_econ_work_refund_cash }
                add_to_variable = { PREV.GO_econ_country_refunds_pending = GO_econ_work_refund }
                add_to_variable = { PREV.GO_econ_country_capital_refunds_pending = GO_econ_work_refund }
                add_to_variable = { PREV.GO_econ_country_debt_retired_pending = GO_econ_work_debt_retired }
                add_to_variable = { PREV.GO_econ_country_refund_cash_pending = GO_econ_work_refund_cash }
                GO_econ_refresh_state_debt = yes
                GO_econ_refresh_state_gdp = yes
            }
            GO_econ_refresh_country_debt = yes
            GO_econ_refresh_project_slots = yes
            GO_econ_refresh_country_gdp = yes
            GO_econ_refresh_fiscal_report = yes
        }
        owner = {
            GO_econ_refresh_country_gdp = yes
            GO_econ_refresh_project_slots = yes
        }
    }
    else_if = {
        limit = {
            check_variable = { GO_econ_state_project_active = 1 }
            has_variable = GO_econ_work_actor
            GO_econ_services_project_has_sponsor = yes
            check_variable = { GO_econ_work_actor = GO_econ_state_project_sponsor }
        }
        # Repairable missing inputs never silently release the paid reservation.
        set_variable = { GO_econ_state_project_refund_pending = 1 }
    }
}

GO_econ_refund_services_project = {
    if = {
        limit = { check_variable = { GO_econ_state_project_type = 1 } }
        GO_econ_refund_development_project = yes
    }
}

GO_econ_cancel_services_project = {
    GO_econ_refund_services_project = yes
}

# STATE. Ownership callbacks may reconcile on the start/work day without giving
# an extra workday. Failed refunds retain the live lock and its retry request.
GO_econ_reconcile_development_project_owner = {
    if = {
        limit = {
            check_variable = { GO_econ_state_project_active = 1 }
            GO_econ_services_project_has_sponsor = yes
        }
        set_variable = { GO_econ_work_actor = GO_econ_state_project_sponsor }
        var:GO_econ_state_project_sponsor = {
            PREV = {
                if = {
                    limit = { NOT = { is_owned_by = PREV } }
                    GO_econ_refund_development_project = yes
                }
            }
        }
    }
}

GO_econ_reconcile_services_project_owner = {
    if = {
        limit = { check_variable = { GO_econ_state_project_type = 1 } }
        GO_econ_reconcile_development_project_owner = yes
    }
}

# STATE. The daily driver calls this once; its day token protects indirect repeats.
# Enemy control pauses without releasing slots. Legal transfer refunds the sponsor.
GO_econ_tick_services_project = {
    GO_econ_reconcile_services_owner = yes
    GO_econ_refresh_services_project_record = yes
    if = {
        limit = {
            GO_econ_services_project_record_valid = yes
            NOT = { check_variable = { GO_econ_state_project_last_work_day = global.num_days } }
        }
        set_variable = { GO_econ_state_project_last_work_day = global.num_days }
        set_variable = { GO_econ_work_actor = GO_econ_state_project_sponsor }
        var:GO_econ_state_project_sponsor = {
            PREV = {
                if = {
                    limit = {
                        OR = {
                            check_variable = { GO_econ_state_project_refund_pending = 1 }
                            NOT = { is_owned_by = PREV }
                        }
                    }
                    GO_econ_refund_services_project = yes
                }
                else_if = {
                    limit = { is_controlled_by = PREV }
                    if = {
                        limit = { check_variable = { GO_econ_state_project_workdays < GO_econ_state_project_required_days } }
                        add_to_variable = { GO_econ_state_project_workdays = 1 }
                        clamp_variable = { var = GO_econ_state_project_workdays max = GO_econ_state_project_required_days }
                    }
                    if = {
                        limit = { check_variable = { GO_econ_state_project_workdays = GO_econ_state_project_required_days } }
                        GO_econ_refresh_state_debt = yes
                        PREV = { GO_econ_refresh_country_debt = yes }
                        set_variable = { GO_econ_work_expected_next_level = GO_econ_state_services_completed_levels }
                        add_to_variable = { GO_econ_work_expected_next_level = 1 }
                        if = {
                            limit = {
                                GO_econ_services_project_finance_ready = yes
                                GO_econ_state_has_services_completed_levels = yes
                                GO_econ_state_has_services_requested_levels = yes
                                has_variable = GO_econ_state_services_funding_owner
                                check_variable = { GO_econ_state_services_funding_owner = PREV }
                                check_variable = { GO_econ_state_project_next_level = GO_econ_work_expected_next_level }
                            }
                            # Paid projects may finish after a status reduction.
                            set_variable = { GO_econ_state_project_active = 0 }
                            set_variable = { GO_econ_state_project_type = 0 }
                            set_variable = { GO_econ_state_project_completed = 1 }
                            set_variable = { GO_econ_state_project_completion_pending = 0 }
                            set_variable = { GO_econ_state_services_completed_levels = GO_econ_state_project_next_level }
                            set_variable = { GO_econ_state_services_completed_levels_populated = 1 }
                            # Add only the new level to the requested funded set;
                            # earlier individually suspended levels stay suspended.
                            add_to_variable = { GO_econ_state_services_requested_levels = 1 }
                            clamp_variable = { var = GO_econ_state_services_requested_levels max = GO_econ_state_services_completed_levels }
                            set_variable = { GO_econ_state_services_requested_levels_populated = 1 }
                            # Existing funded levels and national suspension survive;
                            # the new request operates only at the next settlement.
                            GO_econ_refresh_state_gdp = yes
                            PREV = {
                                GO_econ_refresh_project_slots = yes
                                GO_econ_refresh_country_gdp = yes
                                GO_econ_refresh_fiscal_report = yes
                            }
                        }
                        else = { set_variable = { GO_econ_state_project_completion_pending = 1 } }
                    }
                }
            }
        }
    }
}
