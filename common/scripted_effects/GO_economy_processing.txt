# COUNTRY steel. Transaction-free live capacity rebuild.
GO_econ_refresh_steel_plant_capacity = {
    GO_econ_initialize_country_steel_production = yes
    set_variable = { GO_econ_country_steel_plant_capacity_populated = 0 }
    set_variable = { GO_econ_country_steel_plant_eligible_built = 0 }
    set_variable = { GO_econ_country_steel_plant_assigned = 0 }
    set_variable = { GO_econ_country_steel_plant_missing_assignments = 0 }
    if = {
        limit = { GO_econ_country_steel_production_initialized = yes }
        every_owned_state = {
            limit = { is_controlled_by = PREV }
            if = {
                limit = { GO_econ_state_steel_physical_levels_valid = yes }
                add_to_variable = { PREV.GO_econ_country_steel_plant_eligible_built = building_level@GO_econ_steel_plant }
                if = {
                    limit = {
                        NOT = { GO_econ_state_steel_zero_capacity_unassigned = yes }
                    }
                    if = {
                        limit = { GO_econ_state_has_steel_assignment = yes }
                        add_to_variable = { PREV.GO_econ_country_steel_plant_assigned = GO_econ_state_steel_assigned_levels }
                    }
                    else = { add_to_variable = { PREV.GO_econ_country_steel_plant_missing_assignments = 1 } }
                }
            }
            else = { add_to_variable = { PREV.GO_econ_country_steel_plant_missing_assignments = 1 } }
        }
        if = {
            limit = { check_variable = { GO_econ_country_steel_plant_missing_assignments = 0 } }
            set_variable = { GO_econ_country_steel_plant_capacity_populated = 1 }
            if = {
                limit = { NOT = { GO_econ_country_steel_plant_capacity_ready = yes } }
                set_variable = { GO_econ_country_steel_plant_capacity_populated = 0 }
            }
        }
    }
}

# COUNTRY internal steel preflight preparation. No account writes.
GO_econ_prepare_steel_plant_recipe = {
    set_variable = { GO_econ_work_processing_recipe_id = 1 }
    set_temp_variable = { GO_econ_processing_batches = GO_econ_country_steel_plant_assigned }
    multiply_temp_variable = { var = GO_econ_processing_batches value = global.GO_econ_cfg_steel_plant_batches_per_level }
    set_variable = { GO_econ_work_processing_input_1_quantity = 0 }
    set_variable = { GO_econ_work_processing_input_2_quantity = 0 }
    set_variable = { GO_econ_work_processing_input_3_quantity = 0 }
    set_variable = { GO_econ_work_processing_input_1_quantity = GO_econ_processing_batches }
    multiply_variable = { GO_econ_work_processing_input_1_quantity = global.GO_econ_cfg_steel_plant_input_iron_per_batch }
    set_variable = { GO_econ_work_processing_output_quantity = GO_econ_processing_batches }
    multiply_variable = { GO_econ_work_processing_output_quantity = global.GO_econ_cfg_steel_plant_output_per_batch }
}

# COUNTRY steel. Claim before inspection; a blocked day never retries.
GO_econ_update_steel_plant_production = {
    GO_econ_initialize_processing_config = yes
    GO_econ_initialize_country_steel_production = yes
    if = {
        limit = { GO_econ_steel_plant_daily_update_ready = yes }
        set_variable = { GO_econ_country_steel_plant_last_day = global.num_days }
        set_variable = { GO_econ_country_steel_plant_report_day = global.num_days }
        set_variable = { GO_econ_country_steel_plant_report_accounting_version = 2 }
        set_variable = { GO_econ_country_steel_plant_requirements_populated = 0 }
        set_variable = { GO_econ_country_steel_plant_report_populated = 0 }
        set_variable = { GO_econ_country_steel_plant_report_capacity_populated = 0 }
        set_variable = { GO_econ_country_steel_plant_status = 0 }
        set_variable = { GO_econ_country_steel_plant_report_built = 0 }
        set_variable = { GO_econ_country_steel_plant_report_assigned = 0 }
        set_variable = { GO_econ_country_steel_plant_operating = 0 }
        set_variable = { GO_econ_country_steel_plant_batches_per_level_snapshot = 0 }
        set_variable = { GO_econ_country_steel_plant_output_per_batch_snapshot = 0 }
        set_variable = { GO_econ_country_steel_plant_planned_output = 0 }
        set_variable = { GO_econ_country_steel_plant_output_steel = 0 }
        set_variable = { GO_econ_country_steel_plant_input_iron_per_batch_snapshot = 0 }
        set_variable = { GO_econ_country_steel_plant_required_iron = 0 }
        set_variable = { GO_econ_country_steel_plant_consumed_iron = 0 }
        GO_econ_refresh_steel_plant_capacity = yes
        if = {
            limit = { GO_econ_country_steel_plant_capacity_ready = yes }
            set_variable = { GO_econ_country_steel_plant_report_built = GO_econ_country_steel_plant_eligible_built }
            set_variable = { GO_econ_country_steel_plant_report_assigned = GO_econ_country_steel_plant_assigned }
            set_variable = { GO_econ_country_steel_plant_report_capacity_populated = 1 }
            if = {
                limit = { GO_econ_steel_plant_config_ready = yes }
                GO_econ_prepare_steel_plant_recipe = yes
                if = {
                    limit = {
                        GO_econ_steel_plant_recipe_quantities_ready = yes
                    }
                    set_variable = { GO_econ_country_steel_plant_batches_per_level_snapshot = global.GO_econ_cfg_steel_plant_batches_per_level }
                    set_variable = { GO_econ_country_steel_plant_output_per_batch_snapshot = global.GO_econ_cfg_steel_plant_output_per_batch }
                    set_variable = { GO_econ_country_steel_plant_planned_output = GO_econ_work_processing_output_quantity }
                    set_variable = { GO_econ_country_steel_plant_input_iron_per_batch_snapshot = global.GO_econ_cfg_steel_plant_input_iron_per_batch }
                    set_variable = { GO_econ_country_steel_plant_required_iron = GO_econ_work_processing_input_1_quantity }
                    # Requested requirements are recorded independently of actual allocation.
                    GO_econ_record_processing_demand = yes
                    if = {
                        limit = { check_variable = { GO_econ_work_processing_demand_succeeded = 1 } }
                        set_variable = { GO_econ_country_steel_plant_requirements_populated = 1 }
                        if = {
                            limit = { GO_econ_processing_transaction_accounts_ready = yes }
                            if = {
                                limit = { GO_econ_processing_transaction_ready = yes }
                                GO_econ_execute_processing_transaction = yes
                                if = {
                                    limit = { check_variable = { GO_econ_work_processing_transaction_succeeded = 1 } }
                                    set_variable = { GO_econ_country_steel_plant_operating = GO_econ_country_steel_plant_report_assigned }
                                    set_variable = { GO_econ_country_steel_plant_output_steel = GO_econ_work_processing_output_quantity }
                                    set_variable = { GO_econ_country_steel_plant_consumed_iron = GO_econ_work_processing_input_1_quantity }
                                    set_variable = { GO_econ_country_steel_plant_status = 1 }
                                    set_variable = { GO_econ_country_steel_plant_report_populated = 1 }
                                }
                            }
                            else = {
                                set_variable = { GO_econ_country_steel_plant_status = 2 }
                                set_variable = { GO_econ_country_steel_plant_report_populated = 1 }
                            }
                        }
                    }
                }
            }
        }
    }
}

                    set_variable = { GO_econ_country_microchips_plant_input_titanium_per_batch_snapshot = global.GO_econ_cfg_microchips_plant_input_titanium_per_batch }

                    GO_econ_record_processing_demand = yes

                        limit = { check_variable = { GO_econ_work_processing_demand_succeeded = 1 } }

                            else = {

                                set_variable = { GO_econ_country_microchips_plant_report_populated = 1 }
                            }
                        }
                    }
                }
            }
        }
    }

    GO_econ_initialize_state_processing_assignments = yes

        limit = { GO_econ_can_assign_processing_plant = yes }

            limit = { check_variable = { GO_econ_work_processing_recipe_id = 1 } }
            set_variable = { GO_econ_state_steel_assignment_owner = GO_econ_work_actor }
