# Annual real GDP: billions of the common constant-2027 reference currency.
# Fictional calibration, not measured regional GDP or a verified 2027 forecast.
# Official 2024 national anchors and projection assumptions: docs/HOI4_MOD_DESIGN.md.
# Static state values use the current state-history population shares, rounded to
# 0.001 billion by largest remainder. Native population remains the runtime
# per-capita denominator; this file never writes population or reads factories.
# SOV 2150 / USA 31000 / CHI 20500 / FRA 3450 are editable scenario totals.
# No startup hook lives here. The fresh-start campaign authoring integration alone
# sets GO_econ_2027_startup_authoring = 1 on the requesting COUNTRY before calling.
# Existing saves and partially initialized/authored records are never repaired here.

# STATE, entered directly from the country dataset. GDP/growth inputs only.
GO_econ_author_2027_state_gdp = {
    if = {
        limit = {
            PREV = {
                exists = yes
                has_variable = GO_econ_2027_startup_authoring
                check_variable = { GO_econ_2027_startup_authoring = 1 }
                has_variable = GO_econ_country_2027_gdp_authored_states
            }
            is_owned_by = PREV
            is_controlled_by = PREV
            GO_econ_state_fiscal_initialized = yes
            has_variable = GO_econ_work_2027_starting_gdp
            check_variable = { var = GO_econ_work_2027_starting_gdp value = 0 compare = greater_than_or_equals }
            has_variable = GO_econ_state_gdp
            has_variable = GO_econ_state_gdp_populated
            check_variable = { GO_econ_state_gdp = 0 }
            check_variable = { GO_econ_state_gdp_populated = 0 }
            has_variable = GO_econ_state_growth_modifier_percent
            has_variable = GO_econ_state_growth_modifier_percent_populated
            check_variable = { GO_econ_state_growth_modifier_percent = 0 }
            check_variable = { GO_econ_state_growth_modifier_percent_populated = 0 }
            has_variable = GO_econ_state_last_service_month
            check_variable = { GO_econ_state_last_service_month = -1 }
            has_variable = GO_econ_state_last_growth_month
            check_variable = { GO_econ_state_last_growth_month = -1 }
            has_variable = GO_econ_state_project_active
            check_variable = { GO_econ_state_project_active = 0 }
            NOT = { has_variable = GO_econ_state_project_sponsor }
            NOT = { has_variable = GO_econ_state_2027_gdp_authored }
        }
        set_variable = { GO_econ_state_gdp = GO_econ_work_2027_starting_gdp }
        set_variable = { GO_econ_state_gdp_populated = 1 }
        # Explicit known zero: no additional modifier beyond the approved base rate.
        set_variable = { GO_econ_state_growth_modifier_percent = 0 }
        set_variable = { GO_econ_state_growth_modifier_percent_populated = 1 }
        set_variable = { GO_econ_state_2027_gdp_authored = 1 }
        PREV = { add_to_variable = { GO_econ_country_2027_gdp_authored_states = 1 } }
    }
    clear_variable = GO_econ_work_2027_starting_gdp
}

# COUNTRY. The owner keeps the fresh-start authoring marker live only for setup.
# The result is provenance/coverage metadata, never a second GDP ledger.
GO_econ_author_2027_bookmark_gdp = {
    if = {
        limit = {
            exists = yes
            has_variable = GO_econ_2027_startup_authoring
            check_variable = { GO_econ_2027_startup_authoring = 1 }
            GO_econ_country_fiscal_initialized = yes
            NOT = { has_variable = GO_econ_country_2027_gdp_dataset_complete }
            NOT = { has_variable = GO_econ_country_2027_gdp_authored_states }
            OR = { tag = SOV tag = USA tag = CHI tag = FRA }
        }
        set_variable = { GO_econ_country_2027_gdp_dataset_complete = 0 }
        set_variable = { GO_econ_country_2027_gdp_authored_states = 0 }
        if = {
            limit = { tag = SOV }
            set_variable = { GO_econ_country_2027_gdp_expected_states = 112 }
            GO_econ_author_2027_SOV_gdp = yes
        }
        else_if = {
            limit = { tag = USA }
            set_variable = { GO_econ_country_2027_gdp_expected_states = 52 }
            GO_econ_author_2027_USA_gdp = yes
        }
        else_if = {
            limit = { tag = CHI }
            set_variable = { GO_econ_country_2027_gdp_expected_states = 61 }
            GO_econ_author_2027_CHI_gdp = yes
        }
        else_if = {
            limit = { tag = FRA }
            set_variable = { GO_econ_country_2027_gdp_expected_states = 33 }
            GO_econ_author_2027_FRA_gdp = yes
        }
        if = {
            limit = {
                check_variable = { GO_econ_country_2027_gdp_authored_states = GO_econ_country_2027_gdp_expected_states }
            }
            set_variable = { GO_econ_country_2027_gdp_dataset_complete = 1 }
        }
    }
}
